import requests

def create_customer(data):
    response = requests.post("https://api.example.com/customers", json=data)
    return response.json()

def create_vendor(data):
    response = requests.post("https://api.example.com/vendors", json=data)
    return response.json()

def create_event(data):
    response = requests.post("https://api.example.com/events", json=data)
    return response.json()

def create_job_card(data):
    response = requests.post("https://api.example.com/jobcards", json=data)
    return response.json()

def create_batch(data):
    response = requests.post("https://api.example.com/batches", json=data)
    return response.json()

def create_payment_voucher(data):
    response = requests.post("https://api.example.com/payment-vouchers", json=data)
    payment_voucher_id = response.headers["Location"].split("/")[-1]
    return payment_voucher_id

def create_invoice(data):
    response = requests.post("https://api.example.com/invoices", json=data)
    invoice_id = response.headers["Location"].split("/")[-1]
    return invoice_id
